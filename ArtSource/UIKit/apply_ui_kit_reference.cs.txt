using System.Linq;
using TMPro;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.UI;

// Shared look for every panel: Blue16 kit, one text palette, header plates,
// label margins, icon + amount counters. HUD panels sit on a 1280x720 canvas,
// so they are authored in 1920x1080 units and scaled by 2/3 to match the menus.
const float HudScale = 2f / 3f;
string kit = "Assets/Sprites/UIKit/Blue16/";
Sprite Kit(string n) => AssetDatabase.LoadAssetAtPath<Sprite>(kit + n + ".png");
Sprite SubSprite(string path, string name) => AssetDatabase.LoadAllAssetsAtPath(path).OfType<Sprite>().First(s => name == null || s.name == name);
Sprite gem = SubSprite("Assets/2D Platformer Assets/Graphics/Collectibles/gem.png", "gem_0");
Sprite clover = SubSprite("Assets/Sprites/Collectibles/Clover_Collectible.png", null);
Sprite star = SubSprite("Assets/2D Platformer Assets/Graphics/UI/star.png", null);
Sprite clock = SubSprite("Assets/2D Platformer Assets/Graphics/Overworld/Overworld_tiles_Padded.png", "Overworld_tiles_Padded_57");
Color TextColor = new Color(0.91f, 0.93f, 1f);
Color Accent = new Color(1f, 0.91f, 0.66f);
string report = "";

void Center(RectTransform rt, Vector2 pos, Vector2 size) { rt.anchorMin = rt.anchorMax = rt.pivot = new Vector2(0.5f, 0.5f); rt.anchoredPosition = pos; rt.sizeDelta = size; }
RectTransform Node(string name, Transform parent, Vector2 pos, Vector2 size)
{
    Transform old = parent.Find(name); if (old != null) Object.DestroyImmediate(old.gameObject);
    var go = new GameObject(name, typeof(RectTransform)); go.layer = 5;
    var rt = (RectTransform)go.transform; rt.SetParent(parent, false); Center(rt, pos, size); return rt;
}
Image Sliced(GameObject go, string sprite)
{
    var img = go.GetComponent<Image>() ?? go.AddComponent<Image>();
    img.sprite = Kit(sprite); img.type = Image.Type.Sliced; img.pixelsPerUnitMultiplier = 1.25f; img.color = Color.white; return img;
}
Image Icon(string name, Transform parent, Vector2 pos, float size, Sprite s)
{
    var img = Node(name, parent, pos, new Vector2(size, size)).gameObject.AddComponent<Image>();
    img.sprite = s; img.preserveAspect = true; img.raycastTarget = false; return img;
}
void Text(TextMeshProUGUI t, Color c, float max, TextAlignmentOptions align, bool wrap = false)
{
    if (t == null) return;
    t.color = c; t.enableAutoSizing = true; t.fontSizeMin = 18f; t.fontSizeMax = max; t.fontSize = max;
    t.alignment = align; t.textWrappingMode = wrap ? TextWrappingModes.Normal : TextWrappingModes.NoWrap; t.overflowMode = TextOverflowModes.Ellipsis;
}
void Place(Transform t, Vector2 pos, Vector2 size) { if (t != null) Center((RectTransform)t, pos, size); }
void Button(Transform b, Vector2? pos, Vector2 size, float rightMargin = 28f)
{
    if (b == null) return;
    var rt = (RectTransform)b; if (pos.HasValue) Center(rt, pos.Value, size); else rt.sizeDelta = size;
    var img = Sliced(b.gameObject, "Button_Normal");
    var btn = b.GetComponent<UnityEngine.UI.Button>(); btn.targetGraphic = img; btn.transition = Selectable.Transition.SpriteSwap;
    btn.spriteState = new SpriteState { highlightedSprite = Kit("Button_Focused"), selectedSprite = Kit("Button_Focused"), pressedSprite = Kit("Button_Pressed"), disabledSprite = Kit("Button_Disabled") };
    var anim = b.GetComponent<Animator>(); if (anim != null) Object.DestroyImmediate(anim);
    // The direct label keeps a margin on every side so it never touches the frame.
    foreach (Transform child in b)
    {
        var tmp = child.GetComponent<TextMeshProUGUI>(); if (tmp == null || child.name == "ActiveIndicator") continue;
        var lt = (RectTransform)child; lt.anchorMin = Vector2.zero; lt.anchorMax = Vector2.one; lt.pivot = new Vector2(0.5f, 0.5f);
        lt.offsetMin = new Vector2(28f, 18f); lt.offsetMax = new Vector2(-rightMargin, -12f);
        Text(tmp, TextColor, 44f, TextAlignmentOptions.Center);
    }
}
RectTransform Panel(Transform container, Vector2 size, float scale)
{
    var rt = (RectTransform)container; rt.anchorMin = rt.anchorMax = rt.pivot = new Vector2(0.5f, 0.5f); rt.sizeDelta = size; rt.localScale = Vector3.one * scale;
    Sliced(container.gameObject, "Panel"); return rt;
}
void Header(RectTransform panel, Transform title, float width = 620f)
{
    var h = Node("Header", panel, new Vector2(0f, panel.sizeDelta.y / 2f - 2f), new Vector2(width, 96f)); Sliced(h.gameObject, "Header");
    h.SetAsFirstSibling();
    if (title == null) return;
    title.SetParent(h, false); Center((RectTransform)title, new Vector2(0f, 6f), new Vector2(width - 50f, 66f));
    Text(title.GetComponent<TextMeshProUGUI>(), Accent, 56f, TextAlignmentOptions.Center);
}
RectTransform Box(string name, Transform parent, Vector2 pos, Vector2 size) { var b = Node(name, parent, pos, size); Sliced(b.gameObject, "Slot"); return b; }
void Counter(Transform parent, string iconName, Sprite s, Transform amount, Vector2 pos, float width = 180f)
{
    Icon(iconName, parent, pos, 64f, s);
    if (amount == null) return;
    amount.SetParent(parent, false); Place(amount, pos + new Vector2(48f + width / 2f, 0f), new Vector2(width, 64f));
    Text(amount.GetComponent<TextMeshProUGUI>(), TextColor, 44f, TextAlignmentOptions.Left);
}
void Dim(Transform root) { var img = root.GetComponent<Image>(); if (img != null) { img.sprite = null; img.type = Image.Type.Simple; img.color = new Color(0.03f, 0.04f, 0.12f, 0.7f); } }
